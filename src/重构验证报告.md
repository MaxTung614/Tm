#### 2.2 API客户端配置

```typescript
// 重试配置
interface RetryConfig {
  maxRetries: number;        // 最大重试次数
  retryDelay: number;        // 基础延迟时间
  backoffMultiplier: number; // 退避倍数
  maxDelay: number;          // 最大延迟时间
}

// 默认配置
const DEFAULT_RETRY_CONFIG: RetryConfig = {
  maxRetries: 3,
  retryDelay: 1000,
  backoffMultiplier: 1.5,
  maxDelay: 10000,
};
```
