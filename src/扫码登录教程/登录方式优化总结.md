### 3. **实现短信登录逻辑**

```typescript
// 发送短信验证码
const handleSendSms = async () => {
  if (!smsPhone || !/^1[3-9]\d{9}$/.test(smsPhone)) {
    toast.error('请输入正确的手机号');
    return;
  }
  
  setSmsSending(true);
  try {
    const result = await authService.sendSmsCode(smsPhone);
    
    if (result.success && result.data) {
      setSmsSessionId(result.data.sessionId);
      toast.success(result.message || '验证码已发送');
      
      // 开始60秒倒计时
      setSmsCountdown(60);
      const timer = setInterval(() => {
        setSmsCountdown(prev => {
          if (prev <= 1) {
            clearInterval(timer);
            return 0;
          }
          return prev - 1;
        });
      }, 1000);
    } else {
      throw new Error(result.message || '发送失败');
    }
  } catch (error: any) {
    toast.error(error.message || '发送验证码失败');
  } finally {
    setSmsSending(false);
  }
};

// 短信登录
const handleSmsLogin = async () => {
  if (!smsCode || smsCode.length !== 6) {
    toast.error('请输入6位验证码');
    return;
  }
  
  setSmsLoggingIn(true);
  try {
    const result = await authService.loginWithSms(smsSessionId, smsCode);
    
    if (result.success && result.data) {
      const { cookie, username } = result.data;
      
      // 调用 onSuccess 回调
      onSuccess(cookie, username);
      
      toast.success('登录成功！');
      setIsDialogOpen(false);
      
      // 重置状态
      setSmsPhone('');
      setSmsCode('');
      setSmsSessionId('');
    } else {
      throw new Error(result.message || '登录失败');
    }
  } catch (error: any) {
    toast.error(error.message || '短信登录失败');
  } finally {
    setSmsLoggingIn(false);
  }
};
```

## 🎯 用户名提取策略

所有登录方式都统一使用以下策略提取用户名：

### 优先级：
1. **_nk_** - URL 编码的用户名（最优先）
2. **tracknick** - 备用用户名字段
3. **lgc** - 登录账号（最后备选）

### 代码实现：
```typescript
// 提取用户名
let username = "未知用户";

const nkMatch = cookieString.match(/_nk_=(.*?)($|;)/);
if (nkMatch) {
  try {
    username = decodeURIComponent(nkMatch[1]);
  } catch (err) {
    console.error('解码 _nk_ 失败:', err);
  }
}

if (username === "未知用户") {
  const tracknickMatch = cookieString.match(/tracknick=(.*?)($|;)/);
  if (tracknickMatch) {
    try {
      username = decodeURIComponent(tracknickMatch[1]);
    } catch (err) {
      console.error('解码 tracknick 失败:', err);
    }
  }
}

if (username === "未知用户") {
  const lgcMatch = cookieString.match(/lgc=(.*?)($|;)/);
  if (lgcMatch) {
    try {
      username = decodeURIComponent(lgcMatch[1]);
    } catch (err) {
      console.error('解码 lgc 失败:', err);
    }
  }
}
```

## 🔧 Cookie 验证标准

所有登录方式都会验证以下必要字段：

```typescript
const requiredFields = [
  'cookie2',      // 会话标识
  '_tb_token_',   // 防CSRF token
  '_m_h5_tk',     // H5 API 签名 token
];
```

缺少任何一个字段都会返回错误。
