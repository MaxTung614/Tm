### 1. `/lib/api-services.ts` - 添加短信登录API方法

在文件末尾添加：

```typescript
/**
 * 短信登录相关API
 */
export const smsLoginService = {
  /**
   * 发送短信验证码
   */
  async sendSmsCode(phone: string) {
    const response = await fetch(`${API_BASE_URL}/auth/sms/send`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ phone })
    });
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.message || '发送短信失败');
    }
    
    return await response.json();
  },

  /**
   * 短信验证码登录
   */
  async loginWithSms(sessionId: string, smsCode: string) {
    const response = await fetch(`${API_BASE_URL}/auth/sms/login`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ sessionId, smsCode })
    });
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.message || '短信登录失败');
    }
    
    return await response.json();
  }
};
```

#### 步骤2：在 `Accounts.tsx` 组件内添加短信登录逻辑

在 `loadAccounts` 函数之后，添加：

```typescript
// 发送短信验证码
const handleSendSms = async () => {
  if (!smsPhone || !/^1[3-9]\d{9}$/.test(smsPhone)) {
    toast.error('请输入正确的手机号');
    return;
  }
  
  setSmsSending(true);
  try {
    logInfo('发送短信验证码', { phone: smsPhone });
    
    const result = await smsLoginService.sendSmsCode(smsPhone);
    
    if (result.success && result.data) {
      setSmsSessionId(result.data.sessionId);
      toast.success(result.message || '验证码已发送，请查收短信');
      
      logInfo('短信发送成功', { sessionId: result.data.sessionId });
      
      // 开始60秒倒计时
      setSmsCountdown(60);
      const timer = setInterval(() => {
        setSmsCountdown(prev => {
          if (prev <= 1) {
            clearInterval(timer);
            return 0;
          }
          return prev - 1;
        });
      }, 1000);
    } else {
      throw new Error(result.message || '发送失败');
    }
  } catch (error: any) {
    logError(
      '发送短信验证码失败',
      error,
      ErrorCategory.NETWORK,
      { phone: smsPhone }
    );
    
    const errorId = generateErrorId();
    toast.error(
      createUserFriendlyMessage(error.message || '发送验证码失败', errorId)
    );
  } finally {
    setSmsSending(false);
  }
};

// 短信验证码登录
const handleSmsLogin = async () => {
  if (!smsCode || smsCode.length !== 6) {
    toast.error('请输入6位验证码');
    return;
  }
  
  if (!smsSessionId) {
    toast.error('会话已过期，请重新发送验证码');
    return;
  }
  
  setSmsLoggingIn(true);
  try {
    logInfo('短信登录', { sessionId: smsSessionId });
    
    const result = await smsLoginService.loginWithSms(smsSessionId, smsCode);
    
    if (result.success && result.data) {
      const { cookie, username } = result.data;
      
      logInfo('短信登录成功', { username });
      
      // 保存账号
      const newAccount: Account = {
        id: Date.now().toString(),
        name: username || formData.name.trim() || '未命名账号',
        cookie: cookie,
        createdAt: new Date().toISOString(),
        isActive: true,
      };

      // 调用 API 保存账号
      await accountService.createAccount(newAccount);

      toast.success(`账号 "${newAccount.name}" 添加成功！`);
      
      // 刷新账号列表
      await loadAccounts();
      
      // 关闭对话框并重置状态
      setIsDialogOpen(false);
      setSmsPhone('');
      setSmsCode('');
      setSmsSessionId('');
      setSmsCountdown(0);
      setFormData({ name: '', cookie: '' });
      setEditingAccount(null);
    } else {
      throw new Error(result.message || '登录失败');
    }
  } catch (error: any) {
    logError(
      '短信登录失败',
      error,
      ErrorCategory.AUTH,
      { sessionId: smsSessionId }
    );
    
    const errorId = generateErrorId();
    
    // 特殊处理常见错误
    let errorMessage = error.message;
    if (errorMessage.includes('会话已过期')) {
      toast.error('验证码已过期，请��新发送');
      setSmsSessionId('');
      setSmsCode('');
    } else if (errorMessage.includes('验证码错误') || errorMessage.includes('验证码不正确')) {
      toast.error('验证码错误，请检查后重试');
    } else {
      toast.error(
        createUserFriendlyMessage(errorMessage || '短信登录失败', errorId)
      );
    }
  } finally {
    setSmsLoggingIn(false);
  }
};
```
